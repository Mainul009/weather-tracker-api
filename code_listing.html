 106                              "operator": a["operator"], "threshold": a["threshold"]})

  34  def accuracy(lid):

  55                    "pct_within_2c": {"$round": [

  22  
  23      errors = {}

  34      try:
  35          users.insert_one({"username": u, "email": e.lower(), "password": hashed,
  36                            "role": "user", "tracked_locations": [], "units": "metric"})

  39      return jsonify({"message": "User registered"}), 201

   7  def jwt_required(f):
   8      """Put @jwt_required above a route to make it need a valid login token."""
   9      @wraps(f)
  10      def wrapper(*args, **kwargs):
  11          token = request.headers.get("Authorization", "").removeprefix("Bearer ").strip()
  12          if not token:
  13              return jsonify({"error": "Token is missing"}), 401
  14          if blacklist.find_one({"token": token}):
  15              return jsonify({"error": "Token has been revoked"}), 401
  16          try:
  17              g.user = jwt.decode(token, SECRET_KEY, algorithms=["HS256"])
  18              g.token = token
  19          except jwt.ExpiredSignatureError:
  20              return jsonify({"error": "Token has expired"}), 401
  21          except jwt.InvalidTokenError:
  22              return jsonify({"error": "Token is invalid"}), 401
  23          return f(*args, **kwargs)
  24      return wrapper
  25  
  26  def admin_required(f):
  27      """Like jwt_required, but the user must also be an admin."""
  28      @wraps(f)
  29      @jwt_required
  30      def wrapper(*args, **kwargs):
  31          if g.user.get("role") != "admin":
  32              return jsonify({"error": "Admin access required"}), 403
  33          return f(*args, **kwargs)

  33              if isinstance(v, bool) or not isinstance(v, (int, float)):
  34                  errors[field] = "must be a number"
  35              elif not lo <= v <= hi:
  36                  errors[field] = f"must be between {lo} and {hi}"
  37      if "condition" in data and data["condition"] not in CONDITIONS:
  38          errors["condition"] = f"must be one of {sorted(CONDITIONS)}"
  39      if (not errors and "temp_max" in data and "temp_min" in data
  40              and data["temp_min"] > data["temp_max"]):
  41          errors["temp_min"] = "cannot be greater than temp_max"
  42      return errors
  43  
  44  def validate_observation(data, partial=False):
  45      """Same rules as a forecast, plus an optional comment."""
  46      if not isinstance(data, dict):
  47          return {"body": "must be a JSON object"}
  48      rest = {k: v for k, v in data.items() if k != "comment"}
  49      errors = validate_forecast(rest, partial)
  50      if "comment" in data and (not isinstance(data["comment"], str) or len(data["comment"]) > 500):
  51          errors["comment"] = "must be a string of at most 500 characters"
  52      return errors
  53  
  54  def validate_location(data, partial=False):
  55      if not isinstance(data, dict):
  56          return {"body": "must be a JSON object"}
  57      errors = {}
  58      if not partial:
  59          for k in ("name", "country", "lat", "lng"):
  60              if k not in data:
  61                  errors[k] = "is required"
  62      if "name" in data and (not isinstance(data["name"], str) or not data["name"].strip()):
  63          errors["name"] = "must be a non-empty string"
  64      if "lat" in data and not (isinstance(data["lat"], (int, float)) and -90 <= data["lat"] <= 90):
  65          errors["lat"] = "must be a number between -90 and 90"
  66      if "lng" in data and not (isinstance(data["lng"], (int, float)) and -180 <= data["lng"] <= 180):
  67          errors["lng"] = "must be a number between -180 and 180"
  68      return errors
